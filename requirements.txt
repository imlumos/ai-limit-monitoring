# ============================================================
# Python 依赖清单

# 【怎么用】
#   pip install -r requirements.txt
# 自动安装下面列出的所有库到当前 Python 环境。
#
# 【为什么锁定版本(==)】
#   "PySide6==6.11.1" 表示"必须用这个版本"。
#   锁定版本能保证别人安装时和开发者环境完全一致,
#   避免新版本 API 变动导致项目突然跑不起来。
#
# 【如何升级依赖】
#   1. 改这里的版本号
#   2. pip install -U <包名>
#   3. 充分测试,确认没有 breaking change
# ============================================================

# PySide6 = Qt for Python 官方绑定,提供所有 GUI 控件
# 6.11.1 是当前稳定版,6.x 系列 API 与 5.x 差异较大,不要混用
PySide6==6.11.1

# requests = 最流行的 HTTP 客户端库,用于和各 AI 平台 API 通信
# 注:charset-normalizer 是它的间接依赖(pip 会自动安装),故不在此列出;
# 打包时用 pyinstaller --collect-all charset_normalizer 显式收集即可
requests==2.34.2

# keyring = 跨平台访问系统凭据存储(Windows 凭据管理器 / macOS Keychain)
# 用于"记住 API Key"功能,Key 加密保存,不暴露在文件里
keyring==25.7.0
